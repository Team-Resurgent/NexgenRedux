cmake_minimum_required(VERSION 3.21)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Oldest macOS supported")
project(Nexgen VERSION 0.1 LANGUAGES C CXX)

# One project for every target. The toolchain decides the platform:
#   desktop            the host compiler
#   original Xbox      RXDK's RxdkToolchain.cmake      (XBOXOG)
#   Xbox 360           cmake/Xbox360Toolchain.cmake    (XBOX360) linked into a .xex by RXDK-360's MSBuild steps
#   Xbox One / Series  -DCMAKE_SYSTEM_NAME=WindowsStore          UWP, which also runs on Windows

option(NX_SCRIPT_GENERIC "Use AngelScript's generic calls instead of native ones" OFF)

# The showcase: its scripts, pictures and sounds, in the NexgenRedux-Media repository.
set(NX_MEDIA ${CMAKE_CURRENT_SOURCE_DIR}/../NexgenRedux-Media/media CACHE PATH "Media folder")
get_filename_component(NX_MEDIA "${NX_MEDIA}" ABSOLUTE)
if(NOT EXISTS "${NX_MEDIA}/showcase")
  message(FATAL_ERROR "No media in ${NX_MEDIA}. Run: git submodule update --init")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# RXDK's linker takes a library with no folder in its name to be one of the SDK's.
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Targets with a platform layer and GPU backend so far. Others build the shared layers only.
if(XBOXOG)
  set(NX_PLATFORM xboxog)
elseif(XBOX360)
  set(NX_PLATFORM xbox360)
elseif(CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  set(NX_PLATFORM uwp)
elseif(WIN32)
  set(NX_PLATFORM win32)
  # Nothing to install besides the program itself.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(APPLE)
  set(NX_PLATFORM macos)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(NX_PLATFORM linux)
else()
  set(NX_PLATFORM "")
endif()

include(cmake/Options.cmake)
if(XBOX360)
  include(cmake/Xbox360.cmake)
endif()
include(cmake/AngelScript.cmake)

# --- Layers. A library may only link the ones above it. ---

if(WIN32 OR XBOXOG OR XBOX360)
  set(_thread base/thread_windows.cpp)
elseif(UNIX)
  set(_thread base/thread_posix.cpp)
else()
  set(_thread base/thread_none.cpp)
endif()
add_library(nx_base STATIC ${_thread} base/jobs.cpp base/json.cpp base/json_tree.cpp base/json_value.cpp base/log.cpp base/math.cpp base/memory.cpp base/noise.cpp base/path.cpp)
target_include_directories(nx_base PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
nx_options(nx_base)
if(XBOX360)
  nx_xdk_sources(base/thread_windows.cpp)
endif()

add_library(nx_gpu STATIC gpu/null/null_device.cpp)
target_link_libraries(nx_gpu PUBLIC nx_base)
nx_options(nx_gpu)
if(XBOXOG)
  target_sources(nx_gpu PRIVATE gpu/d3d8/d3d8_device.cpp)
  target_compile_definitions(nx_gpu PUBLIC NX_GPU_D3D8)
elseif(WIN32)
  target_sources(nx_gpu PRIVATE gpu/d3d11/d3d11_device.cpp)
  target_link_libraries(nx_gpu PRIVATE d3d11 dxgi dxguid d3dcompiler)
  if(NX_PLATFORM STREQUAL "uwp")
    target_compile_definitions(nx_gpu PRIVATE NX_UWP)
  endif()
elseif(XBOX360)
  target_sources(nx_gpu PRIVATE gpu/d3d9/d3d9_device.cpp)
  target_compile_definitions(nx_gpu PUBLIC NX_GPU_D3D9)
  nx_xdk_sources(gpu/d3d9/d3d9_device.cpp)
elseif(NX_PLATFORM)
  target_sources(nx_gpu PRIVATE gpu/gl/gl_device.cpp)
endif()

add_library(nx_script STATIC script/binder.cpp script/coroutines.cpp script/script_host.cpp script/bind_math.cpp script/script_abi_test.cpp)
target_link_libraries(nx_script PUBLIC nx_base nx_angelscript)
nx_options(nx_script)

add_library(nx_input STATIC input/input.cpp)
target_link_libraries(nx_input PUBLIC nx_base)
nx_options(nx_input)

# Sound: output is the desktop's through miniaudio, the original Xbox's own, or none yet.
add_library(nx_audio STATIC audio/audio.cpp audio/effects.cpp audio/decoder.cpp audio/third_party.c assets/adpcm.cpp)
target_link_libraries(nx_audio PUBLIC nx_base)
nx_options(nx_audio)
nx_no_warnings(audio/third_party.c)
if(XBOXOG)
  target_sources(nx_audio PRIVATE audio/output_xboxog.cpp)
elseif(XBOX360)
  target_sources(nx_audio PRIVATE audio/output_xbox360.cpp)
  nx_xdk_sources(audio/output_xbox360.cpp)
elseif(WIN32 OR (UNIX AND NOT XBOX360))
  target_sources(nx_audio PRIVATE audio/output_desktop.cpp audio/output_miniaudio.c)
  nx_no_warnings(audio/output_miniaudio.c)
  if(NX_PLATFORM STREQUAL "uwp")
    target_link_libraries(nx_audio PRIVATE mmdevapi ole32)
  elseif(UNIX)
    target_link_libraries(nx_audio PRIVATE pthread ${CMAKE_DL_LIBS} m)
  endif()
endif()

add_library(nx_assets STATIC assets/assets.cpp assets/decode.cpp assets/geometry.cpp assets/import.cpp assets/import_fbx.cpp assets/model.cpp assets/ufbx.c assets/third_party.c)
target_link_libraries(nx_assets PUBLIC nx_audio nx_gpu)
nx_options(nx_assets)
nx_no_warnings(assets/third_party.c)
nx_no_warnings(assets/ufbx.c)
# Too large for the original Xbox, where the cook tool has turned them into something else.
if(NX_PLATFORM AND NOT XBOXOG)
  target_compile_definitions(nx_assets PRIVATE NX_FBX)
endif()

add_library(nx_canvas STATIC canvas/canvas.cpp canvas/look.cpp)
target_link_libraries(nx_canvas PUBLIC nx_assets)
nx_options(nx_canvas)

add_library(nx_scene STATIC scene/physics.cpp scene/space.cpp scene/land.cpp scene/sculpt.cpp scene/shell.cpp scene/cards.cpp scene/nav.cpp scene/mist.cpp scene/voxels.cpp scene/physics3.cpp scene/space_load.cpp scene/prefab.cpp scene/world.cpp scene/paths2d.cpp)
target_link_libraries(nx_scene PUBLIC nx_canvas nx_script)
nx_options(nx_scene)

add_library(nx_ui STATIC ui/draw.cpp ui/keyboard.cpp ui/layout.cpp ui/load.cpp ui/theme.cpp ui/ui.cpp)
target_link_libraries(nx_ui PUBLIC nx_canvas nx_script nx_input)
nx_options(nx_ui)

# TLS, for https. Where there is no network yet, its sockets say so.
file(GLOB _tls CONFIGURE_DEPENDS third_party/mbedtls/library/*.c)
if(XBOX360)
  # The compiler chooses an instruction here that the console does differently, and says so.
  set_source_files_properties(third_party/mbedtls/library/ssl_tls13_generic.c PROPERTIES COMPILE_OPTIONS "-O0")
endif()
add_library(nx_tls STATIC ${_tls})
target_include_directories(nx_tls PUBLIC net/tls third_party/mbedtls/include)
target_compile_definitions(nx_tls PUBLIC MBEDTLS_CONFIG_FILE="nexgen_tls_config.h")
if(XBOX360)
  # Its runtime has no division of numbers twice as wide as a register.
  target_compile_definitions(nx_tls PRIVATE MBEDTLS_NO_UDBL_DIVISION)
endif()
if(MSVC)
  target_compile_options(nx_tls PRIVATE /w)
else()
  target_compile_options(nx_tls PRIVATE -w)
endif()

if(XBOX360)
  set(_socket net/socket_none.cpp)  # until its platform layer is written
else()
  set(_socket net/socket.cpp)
endif()
add_library(nx_net STATIC net/address.cpp net/ftp.cpp net/http.cpp net/meeting.cpp net/net.cpp net/room.cpp net/secure.cpp ${_socket})
target_link_libraries(nx_net PUBLIC nx_base PRIVATE nx_tls)
nx_options(nx_net)
if(WIN32 AND NOT XBOXOG)
  target_link_libraries(nx_net PRIVATE ws2_32 bcrypt)
endif()

if(XBOXOG)
  set(_console platform/xboxog/console_xboxog.cpp)
else()
  set(_console platform/console_none.cpp)
endif()
# What each kind of machine can do that others can't: its own here, and stand-ins for the rest.
set(_machines)
foreach(_machine xbox360 xboxone windows mac linux)
  if(_machine STREQUAL "xbox360" AND XBOX360)
    list(APPEND _machines platform/xbox360/machine_xbox360.cpp)
  elseif(_machine STREQUAL "xboxone" AND NX_PLATFORM STREQUAL "uwp")
    # In nx_platform, which is built with exceptions as the system's calls need.
  elseif(_machine STREQUAL "windows" AND NX_PLATFORM STREQUAL "win32")
    list(APPEND _machines platform/win32/machine_win32.cpp)
  elseif(_machine STREQUAL "mac" AND APPLE)
    list(APPEND _machines platform/macos/machine_macos.mm)
  elseif(_machine STREQUAL "linux" AND NX_PLATFORM STREQUAL "linux")
    list(APPEND _machines platform/linux/machine_linux.cpp)
  else()
    list(APPEND _machines platform/stubs/${_machine}.cpp)
  endif()
endforeach()
add_library(nx_screens STATIC ${_console} ${_machines} bindings/bind_machines.cpp bindings/bind_xbox.cpp bindings/bind_json.cpp bindings/bind_path.cpp bindings/bind_net.cpp bindings/bind_scene.cpp bindings/bind_space.cpp bindings/bind_ui.cpp bindings/bindings.cpp save/save.cpp screens/screens.cpp script/debugger.cpp)
target_link_libraries(nx_screens PUBLIC nx_scene nx_ui nx_input nx_net)
nx_options(nx_screens)
if(NX_PLATFORM STREQUAL "win32")
  target_link_libraries(nx_screens PRIVATE shell32 advapi32)
elseif(APPLE)
  target_link_libraries(nx_screens PRIVATE "-framework AppKit")
elseif(XBOX360)
  nx_xdk_sources(platform/xbox360/machine_xbox360.cpp)
endif()

if(NX_PLATFORM)
  set(_platform platform/${NX_PLATFORM}/platform_${NX_PLATFORM}.cpp)
  if(WIN32)
    list(APPEND _platform platform/windows/platform_windows.cpp)
    if(NX_PLATFORM STREQUAL "uwp")
      list(APPEND _platform platform/uwp/machine_uwp.cpp)
    endif()
  elseif(NX_PLATFORM STREQUAL "linux")
    list(APPEND _platform platform/posix/platform_posix.cpp)
  elseif(NX_PLATFORM STREQUAL "macos")
    set(_platform platform/macos/platform_macos.mm platform/posix/platform_posix.cpp)
  endif()
  add_library(nx_platform STATIC ${_platform})
  target_link_libraries(nx_platform PUBLIC nx_base)
  if(NX_PLATFORM STREQUAL "win32")
    target_link_libraries(nx_platform PRIVATE xinput)
  elseif(NX_PLATFORM STREQUAL "linux")
    target_link_libraries(nx_platform PRIVATE X11 GL)
  elseif(NX_PLATFORM STREQUAL "macos")
    target_link_libraries(nx_platform PRIVATE "-framework Cocoa" "-framework OpenGL" "-framework GameController" "-framework CoreHaptics")
  endif()
  if(NX_PLATFORM STREQUAL "uwp")
    # The system's C++ calls report errors as exceptions, so this library alone has them.
    target_compile_options(nx_platform PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    target_compile_definitions(nx_platform PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  else()
    nx_options(nx_platform)
  endif()

  # Opens every screen unattended when the title starts, for testing on a console.
  option(NX_TOUR "Build the showcase to run its self-test" OFF)

  foreach(_app showcase)
    if(XBOX360)
      # A library here; the title is linked from it by nx_xbox360_title.
      add_library(showcase STATIC app/main.cpp)
      target_link_libraries(showcase PRIVATE nx_screens nx_platform)
    else()
      add_executable(showcase app/main.cpp)
      target_link_libraries(showcase PRIVATE nx_screens nx_platform)
      # Its icon.
      if(NX_PLATFORM STREQUAL "win32")
        target_sources(showcase PRIVATE app/nexgen.rc)
      elseif(APPLE)
        target_sources(showcase PRIVATE app/nexgen.icns)
        set_source_files_properties(app/nexgen.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
      endif()
      if(NOT XBOXOG)
        set_target_properties(showcase PROPERTIES OUTPUT_NAME nexgen)
      endif()
      if(APPLE)
        set_target_properties(showcase PROPERTIES MACOSX_BUNDLE ON MACOSX_BUNDLE_BUNDLE_NAME "Nexgen"
          MACOSX_BUNDLE_GUI_IDENTIFIER com.teamresurgent.nexgen MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION} MACOSX_BUNDLE_ICON_FILE nexgen.icns)
      endif()
      if(NX_PLATFORM STREQUAL "uwp")
        nx_uwp_package(showcase)
      elseif(NOT XBOX360)
        nx_media(showcase ${NX_MEDIA})
      endif()
    endif()
    nx_options(${_app})
    set_target_properties(${_app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${_app})
    if(XBOXOG)
      set(_d "$<$<CONFIG:Debug>:d>")
      target_link_libraries(${_app} PRIVATE libd3d8${_d} libdsound${_d} libxnet${_d} libxapi${_d} libxgraphics${_d} libcpp${_d} libc${_d} libkernel${_d} libcompat${_d})
    endif()
  endforeach()

  if(XBOX360)
    nx_xdk_sources(platform/xbox360/platform_xbox360.cpp)
    nx_xbox360_title(showcase nexgen "Nexgen" 0x4E580003 ${NX_MEDIA})
  endif()
  if(XBOXOG)
    # With NX_COOK set to a built cook tool, the disc holds compressed textures and sounds.
    set(NX_COOK "" CACHE FILEPATH "Host cook tool; empty to ship the media as it is")
    if(NX_COOK)
      set(_media ${CMAKE_BINARY_DIR}/media)
      file(GLOB_RECURSE _media_files CONFIGURE_DEPENDS ${NX_MEDIA}/*)
      add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/media.stamp
        COMMAND ${NX_COOK} --platform xboxog ${NX_MEDIA} ${_media}
        COMMAND ${CMAKE_COMMAND} -E touch ${CMAKE_BINARY_DIR}/media.stamp
        DEPENDS ${_media_files} ${NX_COOK})
      add_custom_target(cooked_media DEPENDS ${CMAKE_BINARY_DIR}/media.stamp)
      add_dependencies(showcase cooked_media)
      file(MAKE_DIRECTORY ${_media})
    else()
      set(_media ${NX_MEDIA})
    endif()
    set(_deploy media=${_media})
    if(NX_TOUR)
      file(WRITE ${CMAKE_BINARY_DIR}/tour/readme.txt "Asks the showcase to run its self-test.")
      list(APPEND _deploy media/tour=${CMAKE_BINARY_DIR}/tour)
    endif()
    rxdk_add_xbe(showcase TITLE_NAME "Nexgen" TITLE_ID 0x4E580002 STACK 262144 ISO DEPLOY ${_deploy})
  endif()
endif()

# --- Tools and tests, on the machine that builds ---

if(NOT XBOXOG AND NOT XBOX360 AND NOT CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  add_executable(cook tools/cook/cook.cpp tools/cook/third_party.c)
  if(WIN32)
    target_sources(cook PRIVATE tools/cook/cook.manifest tools/cook/cook.rc)  # names of files in every language, and an icon
  endif()
  target_link_libraries(cook PRIVATE nx_assets)
  target_compile_definitions(cook PRIVATE NX_VERSION="${PROJECT_VERSION}")
  nx_options(cook)
  nx_no_warnings(tools/cook/third_party.c)

  enable_testing()
  add_executable(nx_tests tests/main.cpp tests/test_base.cpp tests/test_gpu_null.cpp tests/test_audio.cpp tests/test_services.cpp tests/test_script.cpp tests/test_engine.cpp tests/test_scene.cpp tests/test_space.cpp tests/test_ui.cpp tests/test_net.cpp tests/test_ftp.cpp tests/test_json.cpp tests/test_import.cpp)
  target_link_libraries(nx_tests PRIVATE nx_screens)
  target_compile_definitions(nx_tests PRIVATE NX_MEDIA="${NX_MEDIA}")
  nx_options(nx_tests)
  add_test(NAME unit COMMAND nx_tests)
  add_test(NAME self_contained COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckSelfContained.cmake)
  if(NX_PLATFORM)
    # Every screen, many times over: nothing may fail, leak or be read twice.
    add_test(NAME showcase_tour COMMAND showcase --null --tour 20)
  endif()
endif()
